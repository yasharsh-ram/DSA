## Description
You are given an integer array `nums` and two integers, `x` and `k`. You can perform the following operation any number of times (**including zero**):

	- Increase or decrease any element of `nums` by 1.

Return the **minimum** number of operations needed to have **at least** `k` _non-overlapping subarrays_ of size **exactly** `x` in `nums`, where all elements within each subarray are equal.

 

**Example 1:**

**Input:** nums = [5,-2,1,3,7,3,6,4,-1], x = 3, k = 2

**Output:** 8

**Explanation:**

	- Use 3 operations to add 3 to `nums[1]` and use 2 operations to subtract 2 from `nums[3]`. The resulting array is `[5, 1, 1, 1, 7, 3, 6, 4, -1]`.

	- Use 1 operation to add 1 to `nums[5]` and use 2 operations to subtract 2 from `nums[6]`. The resulting array is `[5, 1, 1, 1, 7, 4, 4, 4, -1]`.

	- Now, all elements within each subarray `[1, 1, 1]` (from indices 1 to 3) and `[4, 4, 4]` (from indices 5 to 7) are equal. Since 8 total operations were used, 8 is the output.

**Example 2:**

**Input:** nums = [9,-2,-2,-2,1,5], x = 2, k = 2

**Output:** 3

**Explanation:**

	- Use 3 operations to subtract 3 from `nums[4]`. The resulting array is `[9, -2, -2, -2, -2, 5]`.

	- Now, all elements within each subarray `[-2, -2]` (from indices 1 to 2) and `[-2, -2]` (from indices 3 to 4) are equal. Since 3 operations were used, 3 is the output.

 

**Constraints:**

	- `2 <= nums.length <= 10^5`

	- `-10^6 <= nums[i] <= 10^6`

	- `2 <= x <= nums.length`

	- `1 <= k <= 15`

	- `2 <= k * x <= nums.length`

class Solution {
    multiset<int> left, right;
    long long sumleft=0;
    long long sumright=0;
    void rebalance(){
        int target=(left.size()+right.size()+1)/2;
        while(left.size()>target){
            int val=*left.rbegin();
            left.erase(left.find(val));
            sumleft-=val;
            right.insert(val);
            sumright+=val;
        }
        while(left.size()<target){
            int val=*right.begin();
            right.erase(right.find(val));
            sumright-=val;
            left.insert(val);
            sumleft+=val;
        }
    }
    void add(int num){
        if(left.empty()||num<=*left.rbegin()){
            left.insert(num);
            sumleft+=num;
        }else{
            right.insert(num);
            sumright+=num;
        }
        rebalance();
    }
    void remove(int num){
        auto it=left.find(num);
        if(it!=left.end()){
            sumleft-=num;
            left.erase(it);
        }else{
            it=right.find(num);
            sumright-=num;
            right.erase(it);
        }
        rebalance();
    }
    long long getcost(){
        long long median=*left.rbegin();
        return median*(long long)left.size()-sumleft+sumright-median*(long long)right.size();
    }
public:
    long long minOperations(vector<int>& nums, int x, int k) {
        int n=nums.size();
        vector<long long>cost;
        for(int i=0;i<n;i++){
            add(nums[i]);
            if(i>=x){
                remove(nums[i-x]);
            }
            if(i>=x-1){
                cost.push_back(getcost());
            }
        }
        const long long INF = 1e18;

        vector<vector<long long>> dp(
            n + 1, vector<long long>(k + 1, INF)
        );
        for(int i=0;i<=n;i++){
            dp[i][0]=0;
        }
        for(int i=1;i<=n;i++){
            for(int j=1;j<=k;j++){
                dp[i][j]=dp[i-1][j];
                if(i >=x && dp[i-x][j-1]!=INF){
                    dp[i][j]=min(dp[i][j],dp[i-x][j-1]+cost[i-x]);
                }
            }
        }
        return dp[n][k];
    }
};
