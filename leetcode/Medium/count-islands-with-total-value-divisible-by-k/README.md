## Description
You are given an `m x n` matrix `grid` and a positive integer `k`. An **island** is a group of **positive** integers (representing land) that are **4-directionally** connected (horizontally or vertically).

The **total value** of an island is the sum of the values of all cells in the island.

Return the number of islands with a total value **divisible by** `k`.

**Output:** 6

**Explanation:**

The grid contains six islands, each with a total value that is divisible by 3.

 

**Constraints:**

	- `m == grid.length`

	- `1 <= m, n <= 1000`

	- `0 <= grid[i][j] <= 10^6`

	- `1 <= k <= 10^6`

class Solution {
      long long dfs(int x,int y,int m,int n,vector<vector<int>>& grid,vector<vector<bool>>& visit){
        if(x<0||x>=m||y<0||y>=n)return 0;
        if(grid[x][y]==0||visit[x][y])return 0;
        visit[x][y]=true;
        long long sum=grid[x][y];
        vector<int>dirx={0,1,0,-1};
        vector<int>diry={-1,0,1,0};
        for(int i=0;i<4;i++){
            int r=x+dirx[i];
            int c=y+diry[i];
            sum+=dfs(r,c,m,n,grid,visit);
        }
        return sum;
      }
public:
    int countIslands(vector<vector<int>>& grid, int k) {
        int m=grid.size();
        int n=grid[0].size();
        vector<vector<bool>>visit(m,vector<bool>(n,false));
        int count=0;
        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++){
                if(grid[i][j]>0&&!visit[i][j]){
                    long long sum=dfs(i,j,m,n,grid,visit);
                    if(sum%k==0){
                    count++;
                    }
                }
            }
        }
        return count;
    }
};
